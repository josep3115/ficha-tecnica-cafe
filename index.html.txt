    <!-- Application Script -->
    <script>
        // State Keys corresponding to inputs and view elements
        const fields = [
            'name', 'producer', 'score', 'notes',
            'proceso', 'rueda_p', 'rueda_s', 'residual',
            'sensacion', 'dulzor', 'acidez', 'intensidad',
            'origen', 'altura', 'tueste'
        ];

        let qrCodeObj = null;

        // Initialize Lucide Icons
        document.addEventListener('DOMContentLoaded', () => {
            lucide.createIcons();
            initQR();
            loadStateFromURL();
            bindEvents();
        });

        // Initialize QRCode library instance
        function initQR() {
            const qrElement = document.getElementById("qrcode");
            qrElement.innerHTML = "";
            qrCodeObj = new QRCode(qrElement, {
                text: window.location.href,
                width: 128,
                height: 128,
                colorDark: "#0e0e10",
                colorLight: "#ffffff",
                correctLevel: QRCode.CorrectLevel.M
            });
        }

        // Read form inputs and sync to Mobile UI Preview
        function syncPreview() {
            const data = {};

            fields.forEach(field => {
                const inputVal = document.getElementById(`in_${field}`).value;
                const viewElem = document.getElementById(`view_${field}`);
                
                if (viewElem) {
                    viewElem.textContent = inputVal || '—';
                }
                data[field] = inputVal;
            });

            // Update URL parameters dynamically without reloading page
            updateURLParams(data);

            // Trigger QR update
            updateQRCode();
        }

        // Update URL search parameters
        function updateURLParams(data) {
            const params = new URLSearchParams();
            Object.keys(data).forEach(key => {
                if (data[key]) {
                    params.set(key, data[key]);
                }
            });
            const newURL = `${window.location.pathname}?${params.toString()}`;
            window.history.replaceState({}, '', newURL);
        }

        // Update QR Code graphic and trigger animation
        function updateQRCode() {
            if (!qrCodeObj) return;

            const currentURL = window.location.href;
            qrCodeObj.clear();
            qrCodeObj.makeCode(currentURL);

            // Animate QR container to notify visual update
            const wrapper = document.getElementById('qrContainerWrapper');
            wrapper.classList.remove('qr-updated');
            void wrapper.offsetWidth; // Trigger reflow
            wrapper.classList.add('qr-updated');
        }

        // Load parameters from URL if present
        function loadStateFromURL() {
            const params = new URLSearchParams(window.location.search);
            let hasParams = false;

            fields.forEach(field => {
                if (params.has(field)) {
                    hasParams = true;
                    const val = params.get(field);
                    const inputElem = document.getElementById(`in_${field}`);
                    if (inputElem) {
                        inputElem.value = val;
                    }
                }
            });

            // Initial Sync
            syncPreview();
        }

        // Attach input listeners
        function bindEvents() {
            fields.forEach(field => {
                const inputElem = document.getElementById(`in_${field}`);
                if (inputElem) {
                    inputElem.addEventListener('input', syncPreview);
                }
            });

            // Reset Button
            document.getElementById('btnReset').addEventListener('click', () => {
                const defaultValues = {
                    name: 'SL28',
                    producer: 'Wilmer Cuellar · Finca Las Brisas',
                    score: '88,00',
                    notes: 'Avinado • Mora • Nibs de Cacao',
                    proceso: 'Natural',
                    rueda_p: 'Ácido Fermentado',
                    rueda_s: 'Alcohol Fermentado',
                    residual: 'Cacao',
                    sensacion: 'Sedosa',
                    dulzor: '10 / 15 · Medio',
                    acidez: '11 / 15 · Alta',
                    intensidad: '11 / 15 · Alta',
                    origen: 'Acevedo, Huila',
                    altura: '1700 msnm',
                    tueste: '2026'
                };

                Object.keys(defaultValues).forEach(key => {
                    const el = document.getElementById(`in_${key}`);
                    if (el) el.value = defaultValues[key];
                });

                syncPreview();
                showToast('Valores restablecidos por defecto');
            });

            // Share / Copy Link Button
            document.getElementById('btnShare').addEventListener('click', () => {
                const dummy = document.createElement('input');
                document.body.appendChild(dummy);
                dummy.value = window.location.href;
                dummy.select();
                document.execCommand('copy');
                document.body.removeChild(dummy);

                showToast('¡Enlace con esta ficha copiado!');
            });

            // Download QR code image
            document.getElementById('btnDownloadQR').addEventListener('click', () => {
                const qrImage = document.querySelector('#qrcode img');
                const canvas = document.querySelector('#qrcode canvas');
                
                let imageSrc = '';
                if (qrImage && qrImage.src) {
                    imageSrc = qrImage.src;
                } else if (canvas) {
                    imageSrc = canvas.toDataURL("image/png");
                }

                if (imageSrc) {
                    const downloadLink = document.createElement("a");
                    const coffeeName = document.getElementById('in_name').value || 'cafe';
                    downloadLink.href = imageSrc;
                    downloadLink.download = `QR_Cata_${coffeeName.replace(/\s+/g, '_')}.png`;
                    document.body.appendChild(downloadLink);
                    downloadLink.click();
                    document.body.removeChild(downloadLink);
                    showToast('Código QR descargado');
                }
            });
        }

        // Helper Toast Notification
        function showToast(msg) {
            const toast = document.getElementById('toast');
            const toastMsg = document.getElementById('toastMsg');
            toastMsg.textContent = msg;
            
            toast.classList.remove('translate-y-20', 'opacity-0');
            toast.classList.add('translate-y-0', 'opacity-100');

            setTimeout(() => {
                toast.classList.remove('translate-y-0', 'opacity-100');
                toast.classList.add('translate-y-20', 'opacity-0');
            }, 3000);
        }
    </script>
